function Test-Command($name) {
    $cmd = Get-Command $name -ErrorAction SilentlyContinue
    if (-not $cmd) {
        Write-Host "FEHLT: $name" -ForegroundColor Red
        return $false
    }
    Write-Host "OK: $name -> $($cmd.Source)" -ForegroundColor Green
    return $true

Write-Host "--- Voraussetzungen ---" -ForegroundColor Yellow
$ok = $true
$ok = (Test-Command "rustc") -and $ok
$ok = (Test-Command "cargo") -and $ok
$ok = (Test-Command "node") -and $ok
$ok = (Test-Command "npm") -and $ok

    Write-Host "OK: WiX Toolset gefunden" -ForegroundColor Green

    Write-Host "WARNUNG: WiX Toolset nicht gefunden – MSI-Build könnte fehlschlagen." -ForegroundColor Yellow
